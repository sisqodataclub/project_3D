server {
    listen 80;
    server_name franciscodes.com www.franciscodes.com;

    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name franciscodes.com www.franciscodes.com;

    ssl_certificate /etc/letsencrypt/live/franciscodes.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/franciscodes.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    location / {
        proxy_pass http://franciscodes-backend:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /static/ {
        alias /app/static/;
    }

    location / {
        proxy_pass http://franciscodes-frontend:80;
    }
}
